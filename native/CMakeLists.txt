function(shader_dependency name archive upstream sha256)
    set(local "${CMAKE_CURRENT_SOURCE_DIR}/../build/downloads/${archive}.tar.gz")
    if(EXISTS "${local}")
        set(upstream "${local}")
    endif()
    FetchContent_Declare(${name} URL "${upstream}" URL_HASH "SHA256=${sha256}" EXCLUDE_FROM_ALL ${ARGN})
    FetchContent_MakeAvailable(${name})
    set(${name}_SOURCE_DIR "${${name}_SOURCE_DIR}" PARENT_SCOPE)
    set(${name}_BINARY_DIR "${${name}_BINARY_DIR}" PARENT_SCOPE)
endfunction()
